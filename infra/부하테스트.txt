# 부하 테스트 세부 환경 설정
- 저렴한 환경 부하 테스트 : 문제 탐지(Probe)
- 운영 환경 부하 테스트 : 수치 확정, 운영 유사(Prodlike)

# 컨테이너별 자원 제한

## Probe(문제 탐지용) 자원 제한
- mysql: cpus: 0.50, memory: 768M (스왑 0)
- prometheus: cpus: 0.25, memory: 512M
- grafana: cpus: 0.25, memory: 512M
- k6: cpus: 0.50, memory: 512M
- Probe 총합(대략): CPU 1.5 / RAM 2.3GB

## Prodlike(운영 유사) 자원 제한
- mysql: cpus: 1.50, memory: 3G
- prometheus: cpus: 0.75, memory: 1.5G
- grafana: cpus: 0.50, memory: 1G
- k6: cpus: 1.00, memory: 1G
- Prodlike 총합(대략): CPU 3.75 / RAM 6.5GB

# 1. make 명령어 사용, 2. 셸 명령어 사용
## 1. make 명령어 확인
```
#infra 폴더 이동 ex) C:\Users\qkrqh\IdeaProjects\NBE8-10-2-Team03\infra
cd infra

#make 명령어 확인
make

#타이핑 시 출력
Usage:
  make up            # 기본 실행 (자원 제한 없음)
  make k6            # 기본 k6 실행

  make probe-up      # probe(문제 탐지) 실행
  make probe-k6      # probe k6 실행

  make prodlike-up   # prodlike(운영 유사) 실행
  make prodlike-k6   # prodlike k6 실행

  make down          # 컨테이너 종료
  make inspect       # mysql 자원 제한(raw) 확인
  make limits        # 전체 컨테이너 CPU/MEM 제한 확인

```
## 2. spring project terminal 시작 기준 셸 명령어
```
#infra 폴더 이동 ex) C:\Users\qkrqh\IdeaProjects\NBE8-10-2-Team03\infra
cd infra
```
```
#자원 제한 없이 실행

#mysql, prometheus, grafana 실행 (최초 1회, 서버 재시작 시)
docker compose up -d mysql prometheus grafana

#k6 script 실행 (필요할 때 실행, 부하 테스트 돌리고 싶을 때 마다)
docker compose --profile k6 run --rm k6
```

```
# probe(문제 탐지) 실행

#mysql, prometheus, grafana 실행 (최초 1회, 서버 재시작 시)
docker compose -f docker-compose.yml -f docker-compose.probe.yml up -d mysql prometheus grafana

#k6 script 실행 (필요할 때 실행, 부하 테스트 돌리고 싶을 때 마다)
docker compose -f docker-compose.yml -f docker-compose.probe.yml --profile k6 run --rm k6
```

```
# prodlike(운영 유사) 실행

#mysql, prometheus, grafana 실행 (최초 1회, 서버 재시작 시)
docker compose -f docker-compose.yml -f docker-compose.prodlike.yml up -d mysql prometheus grafana

#k6 script 실행 (필요할 때 실행, 부하 테스트 돌리고 싶을 때 마다)
docker compose -f docker-compose.yml -f docker-compose.prodlike.yml --profile k6 run --rm k6
```

```
#켜진 컨테이너 끄기
docker compose down
```
